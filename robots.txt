import { MetadataRoute } from 'next'

export default function robots(): MetadataRoute.Robots {
  // Domínio de Produção Oficial
  const SITE_URL = 'https://concursosmaranhao.com.br'

  return {
    rules: {
      userAgent: '*',
      allow: '/',
      disallow: ['/admin/', '/api/'],
    },
    sitemap: `${SITE_URL}/sitemap.xml`,
  }
}
